Quick Start

main.dart 
import 'package:nepal_admin_api/nepal_admin_api.dart';

void main() {
  // Get all provinces
  final provinces = NepalAdminApi.getAllProvinces();
  print('Provinces: ${provinces.length}');

  // Get province by ISO code
  final bagmati = NepalAdminApi.getProvinceById('NP-P3');
  print('Bagmati districts: ${bagmati?.districts.length}');

  // Search local bodies
  final result = NepalAdminApi.search('Bharatpur');
  for (final lb in result.localBodies) {
    print('${lb.name} (${lb.typeName}) - ${lb.districtName}, ${lb.provinceName}');
    if (lb.cbsCode != null) {
      print('  CBS Code: ${lb.cbsCode} (verified: ${lb.cbsCodeVerified})');
    }
  }

  // National stats
  final stats = NepalAdminApi.getNationalStats();
  print('Total wards: ${stats.wards}');
}
